    <!-- JAVASCRIPT XỬ LÝ LOGIC & API GOOGLE SHEETS -->
    <script>
        // Đường dẫn API Google Apps Script của bạn
        const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyiqYziypX77tHmYPHF2o_ODWNTzge7PTGe0Lm3fO2722sWTxaf9vgaTlj-3-7TJRoTwQ/exec";

        let editingId = null; // Biến đánh dấu đang sửa phiếu nào

        // Chuyển tab điều hướng
        function switchTab(tab) {
            document.getElementById('tabCreate').classList.add('hidden');
            document.getElementById('tabMiddle').classList.add('hidden');
            document.getElementById('tabHistory').classList.add('hidden');

            document.getElementById('navCreate').className = "flex flex-col items-center text-gray-400 transition-all";
            document.getElementById('navMiddle').className = "flex flex-col items-center text-gray-400 transition-all";
            document.getElementById('navHistory').className = "flex flex-col items-center text-gray-400 transition-all";

            if (tab === 'create') {
                document.getElementById('tabCreate').classList.remove('hidden');
                document.getElementById('navCreate').className = "flex flex-col items-center text-blue-600 transition-all";
            } else if (tab === 'middle') {
                document.getElementById('tabMiddle').classList.remove('hidden');
                document.getElementById('navMiddle').className = "flex flex-col items-center text-blue-600 transition-all";
            } else if (tab === 'history') {
                document.getElementById('tabHistory').classList.remove('hidden');
                document.getElementById('navHistory').className = "flex flex-col items-center text-blue-600 transition-all";
                loadHistory();
            }
        }

        // Chuyển từ Bước 1 sang Bước 2
        function goToStep2() {
            let name = document.getElementById('customerName').value.trim();
            let vehicle = document.getElementById('vehicleInfo').value.trim();

            if (!name || !vehicle) {
                alert("⚠️ Vui lòng nhập đầy đủ tên chủ xe và thông tin xe!");
                return;
            }

            document.getElementById('displayCustomerInfo').innerText = `${name} - ${vehicle}`;
            document.getElementById('step1').classList.add('hidden');
            document.getElementById('step2').classList.remove('hidden');

            if (document.getElementById('itemsList').children.length === 0) {
                addItemRow(); // Thêm sẵn 1 dòng khi mở bảng
            }
        }

        function backToStep1() {
            document.getElementById('step2').classList.add('hidden');
            document.getElementById('step1').classList.remove('hidden');
        }

        // Thêm một dòng phụ tùng mới
        function addItemRow(name = "", qty = 1, price = 0) {
            let container = document.getElementById('itemsList');
            let rowId = 'row_' + Date.now() + Math.random().toString(36.substr(2, 5));
            
            let div = document.createElement('div');
            div.className = "grid grid-cols-12 gap-1.5 items-center bg-gray-50 p-2 rounded-xl border border-gray-100 item-row";
            div.id = rowId;
            div.innerHTML = `
                <div class="col-span-4">
                    <input type="text" placeholder="Tên phụ tùng" value="${name}" class="item-name w-full px-2 py-1.5 rounded-lg border border-gray-200 text-xs bg-white focus:outline-none focus:ring-1 focus:ring-blue-500">
                </div>
                <div class="col-span-2">
                    <input type="number" min="1" value="${qty}" oninput="calculateTotal()" class="item-qty w-full px-1 py-1.5 rounded-lg border border-gray-200 text-xs bg-white text-center focus:outline-none focus:ring-1 focus:ring-blue-500">
                </div>
                <div class="col-span-3">
                    <input type="number" min="0" value="${price}" oninput="calculateTotal()" placeholder="Đơn giá" class="item-price w-full px-2 py-1.5 rounded-lg border border-gray-200 text-xs bg-white text-right focus:outline-none focus:ring-1 focus:ring-blue-500">
                </div>
                <div class="col-span-3 flex items-center justify-between pl-1">
                    <span class="item-total text-xs font-bold text-gray-700 text-right w-full">0đ</span>
                    <button onclick="document.getElementById('${rowId}').remove(); calculateTotal();" class="text-red-400 hover:text-red-600 ml-1.5 px-1">
                        <i class="fa-solid fa-trash-can text-xs"></i>
                    </button>
                </div>
            `;
            container.appendChild(div);
            calculateTotal();
        }

        // Tự động tính tiền từng dòng và tổng tiền toàn bộ
        function calculateTotal() {
            let rows = document.querySelectorAll('.item-row');
            let grandTotal = 0;

            rows.forEach(row => {
                let qty = parseFloat(row.querySelector('.item-qty').value) || 0;
                let price = parseFloat(row.querySelector('.item-price').value) || 0;
                let total = qty * price;
                
                row.querySelector('.item-total').innerText = total.toLocaleString('vi-VN') + 'đ';
                grandTotal += total;
            });

            document.getElementById('grandTotal').innerText = grandTotal.toLocaleString('vi-VN') + ' đ';
        }

        // Thu thập dữ liệu các dòng phụ tùng
        function getItemsData() {
            let rows = document.querySelectorAll('.item-row');
            let items = [];
            rows.forEach(row => {
                items.push({
                    name: row.querySelector('.item-name').value.trim() || "Linh kiện",
                    qty: parseFloat(row.querySelector('.item-qty').value) || 1,
                    price: parseFloat(row.querySelector('.item-price').value) || 0
                });
            });
            return items;
        }

        // Lưu giao dịch (Tạo mới hoặc Cập nhật) lên Google Sheets
        function saveTransaction() {
            let customerName = document.getElementById('customerName').value.trim();
            let vehicleInfo = document.getElementById('vehicleInfo').value.trim();
            let items = getItemsData();

            if (items.length === 0) {
                alert("⚠️ Vui lòng thêm ít nhất một phụ tùng hoặc dịch vụ!");
                return;
            }

            let grandTotalText = document.getElementById('grandTotal').innerText;

            let payload = {
                action: editingId ? "update" : "create",
                id: editingId,
                customerName: customerName,
                vehicleInfo: vehicleInfo,
                items: items,
                totalAmount: grandTotalText
            };

            let saveBtn = document.getElementById('saveBtn');
            saveBtn.disabled = true;
            saveBtn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i>Đang lưu lên Google Sheet...`;

            fetch(SCRIPT_URL, {
                method: "POST",
                body: JSON.stringify(payload)
            })
            .then(res => res.json())
            .then(data => {
                saveBtn.disabled = false;
                saveBtn.innerHTML = `<i class="fa-solid fa-floppy-disk mr-1"></i>Lưu Giao Dịch & Đẩy Lên Sheet`;
                
                if (data.status === "success") {
                    alert(editingId ? "✅ Cập nhật phiếu thành công!" : "🎉 Lưu giao dịch lên Google Sheets thành công!");
                    resetForm();
                } else {
                    alert("❌ Lỗi: " + data.message);
                }
            })
            .catch(err => {
                saveBtn.disabled = false;
                saveBtn.innerHTML = `<i class="fa-solid fa-floppy-disk mr-1"></i>Lưu Giao Dịch & Đẩy Lên Sheet`;
                alert("❌ Lỗi kết nối mạng hoặc Apps Script!");
                console.error(err);
            });
        }

        // Reset form về trạng thái tạo mới ban đầu
        function resetForm() {
            editingId = null;
            document.getElementById('customerName').value = "";
            document.getElementById('vehicleInfo').value = "";
            document.getElementById('itemsList').innerHTML = "";
            document.getElementById('step2').classList.add('hidden');
            document.getElementById('step1').classList.remove('hidden');
            document.getElementById('grandTotal').innerText = "0 đ";
        }

        // Tải danh sách lịch sử từ Google Sheets
        function loadHistory() {
            let loading = document.getElementById('historyLoading');
            let listContainer = document.getElementById('historyList');
            
            loading.style.display = "block";
            listContainer.innerHTML = "";

            fetch(SCRIPT_URL, {
                method: "POST",
                body: JSON.stringify({ action: "getHistory" })
            })
            .then(res => res.json())
            .then(data => {
                loading.style.display = "none";
                if (data.status === "success") {
                    let records = data.data;
                    if (records.length === 0) {
                        listContainer.innerHTML = `<div class="text-center py-8 text-gray-400 text-xs">Chưa có giao dịch nào được lưu.</div>`;
                        return;
                    }

                    records.forEach(rec => {
                        let card = document.createElement('div');
                        card.className = "ios-card p-4 space-y-2 border border-gray-100";
                        
                        let itemsHtml = rec.items.map(i => `<li class="text-xs text-gray-600">• ${i.name} (SL: ${i.qty}) - ${i.price.toLocaleString('vi-VN')}đ</li>`).join('');

                        card.innerHTML = `
                            <div class="flex justify-between items-start">
                                <div>
                                    <h3 class="font-bold text-sm text-gray-800">${rec.customerName}</h3>
                                    <p class="text-xs text-blue-600 font-medium">${rec.vehicleInfo}</p>
                                </div>
                                <span class="text-xs text-gray-400">${rec.date}</span>
                            </div>
                            <ul class="pl-2 space-y-0.5 border-t border-b border-gray-100 py-1.5 my-1">
                                ${itemsHtml}
                            </ul>
                            <div class="flex justify-between items-center pt-1">
                                <span class="text-xs font-bold text-red-600">Tổng: ${rec.totalAmount}</span>
                                <div class="space-x-2">
                                    <button onclick='editTransaction(${JSON.stringify(JSON.stringify(rec))})' class="bg-blue-50 text-blue-600 hover:bg-blue-100 px-3 py-1 rounded-lg text-xs font-semibold transition-all">
                                        <i class="fa-solid fa-pen-to-square mr-1"></i>Sửa
                                    </button>
                                    <button onclick="deleteTransaction('${rec.id}')" class="bg-red-50 text-red-600 hover:bg-red-100 px-3 py-1 rounded-lg text-xs font-semibold transition-all">
                                        <i class="fa-solid fa-trash mr-1"></i>Xóa
                                    </button>
                                </div>
                            </div>
                        `;
                        listContainer.appendChild(card);
                    });
                } else {
                    listContainer.innerHTML = `<div class="text-center py-8 text-red-500 text-xs">Không thể tải dữ liệu lịch sử.</div>`;
                }
            })
            .catch(err => {
                loading.style.display = "none";
                listContainer.innerHTML = `<div class="text-center py-8 text-red-500 text-xs">Lỗi kết nối khi tải lịch sử.</div>`;
                console.error(err);
            });
        }

        // Chỉnh sửa phiếu từ lịch sử
        function editTransaction(recStr) {
            let rec = JSON.parse(recStr);
            editingId = rec.id;

            document.getElementById('customerName').value = rec.customerName;
            document.getElementById('vehicleInfo').value = rec.vehicleInfo;
            
            let container = document.getElementById('itemsList');
            container.innerHTML = "";
            rec.items.forEach(i => {
                addItemRow(i.name, i.qty, i.price);
            });

            switchTab('create');
            goToStep2();
            document.getElementById('saveBtn').innerHTML = `<i class="fa-solid fa-floppy-disk mr-1"></i>Cập Nhật Lại Phiếu`;
        }

        // Xóa phiếu khỏi Google Sheets
        function deleteTransaction(id) {
            if (!confirm("Bạn có chắc chắn muốn xóa phiếu này không?")) return;

            fetch(SCRIPT_URL, {
                method: "POST",
                body: JSON.stringify({ action: "delete", id: id })
            })
            .then(res => res.json())
            .then(data => {
                if (data.status === "success") {
                    alert("🗑️ Đã xóa phiếu thành công!");
                    loadHistory();
                } else {
                    alert("❌ Lỗi khi xóa: " + data.message);
                }
            })
            .catch(err => {
                alert("❌ Lỗi kết nối!");
                console.error(err);
            });
        }
    </script>
